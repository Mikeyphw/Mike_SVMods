using System;

using WildEggsHatching.Management;

namespace WildEggsHatching.DesertTomb.Calculation
{
    /// <summary>
    /// Refactored version of DifficultyManager using the new centralized system
    /// </summary>
    public class DifficultyManager
    {
        private readonly SpawnConfiguration _config;
        private readonly TextOptionValues.TombDifficulty _tombDifficulty;
        private readonly SpawnRateCalculator _calculator;

        public double MineMultiplier => _config.Multipliers.MineMultiplier;
        public double ChanceToDoubleDropItem => _config.Multipliers.ChanceToDoubleDropItem;
        public double DropItemChanceMultiplier => _config.Multipliers.DropItemChanceMultiplier;
        public double DropItemChanceMultiplierTheTombPharaoh =>
            _config.Multipliers.DropItemChanceMultiplierTheTombPharaoh;

        private static double ReduceMultiplierIfNecessary(double multiplier) =>
            ProgressionManager.HasScrollOrCompletedQuest
                ? multiplier
                    * SpawnConfiguration.Default.Multipliers.QuestCompletionDifficultyReduction
                : multiplier;

        public DifficultyManager(TextOptionValues.TombDifficulty tombDifficulty)
        {
            _tombDifficulty = tombDifficulty;
            _config = SpawnConfiguration.Default;
            _calculator = new SpawnRateCalculator(_config);

            ApplyDifficultyToMultipliers();
        }

        private void ApplyDifficultyToMultipliers()
        {
            var difficultyConfig = _config.DifficultyConfigurations.ContainsKey(_tombDifficulty)
                ? _config.DifficultyConfigurations[_tombDifficulty]
                : new DifficultyConfiguration();

            // Note: These are now read-only properties that use configuration values
            // The actual values are now configured in SpawnConfiguration
        }

        /// <summary>
        /// Get the difficulty multiplier for a specific value
        /// </summary>
        public double GetDifficultyMultiplier(double baseValue, double modifier)
        {
            return _calculator.CalculateDifficultyMultiplier(_tombDifficulty, baseValue, modifier);
        }

        /// <summary>
        /// Get the maximum number of monsters allowed per tomb
        /// </summary>
        public static Dictionary<TombMonsterType, int> GetMaxMonsterInTomb()
        {
            return new Dictionary<TombMonsterType, int>()
            {
                { TombMonsterType.AssassinBug, 5 },
                { TombMonsterType.Bat, 2 },
                { TombMonsterType.BruteMummy, 6 },
                { TombMonsterType.Duggy, 5 },
                { TombMonsterType.RoyalSerpent, 1 },
                { TombMonsterType.HauntedSkull, 1 },
                { TombMonsterType.Ghost, 2 },
                { TombMonsterType.Mummy, 50 },
                { TombMonsterType.Skeleton, 6 },
                { TombMonsterType.SkeletonMage, 3 },
                { TombMonsterType.Spider, 2 },
                { TombMonsterType.TheTombPharaoh, 1 },
            };
        }

        /// <summary>
        /// Get the chance for barrels to appear instead of plants
        /// </summary>
        public double GetBarrelPlantChance()
        {
            return _config.Global.BarrelPlantChance;
        }

        /// <summary>
        /// Validate configuration values
        /// </summary>
        public bool ValidateConfiguration()
        {
            return MineMultiplier > 0
                && DropItemChanceMultiplier > 0
                && DropItemChanceMultiplierTheTombPharaoh > 0;
        }
    }
}
